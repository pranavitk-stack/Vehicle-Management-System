    class Customer {
        -int customerId
        -String name
        -String phone
        -String email
        -String address
        -Date createdAt
        +addCustomer() boolean
        +searchCustomer(id: int) Customer
        +updateCustomer() boolean
        +deleteCustomer(id: int) boolean
        +getAllCustomers() List~Customer~
        +toString() String
    }

    class Vehicle {
        -int vehicleId
        -int customerId
        -String registrationNumber
        -String brand
        -String model
        -String vehicleType
        -int year
        +addVehicle() boolean
        +getVehiclesByCustomer(customerId: int) List~Vehicle~
        +searchByRegNumber(regNo: String) Vehicle
        +updateVehicle() boolean
        +getAllVehicles() List~Vehicle~
        +toString() String
    }

    class Appointment {
        -int appointmentId
        -int customerId
        -int vehicleId
        -Date appointmentDate
        -String issueDescription
        -String status
        +bookAppointment() boolean
        +cancelAppointment(id: int) boolean
        +updateStatus(id: int, status: String) boolean
        +getAppointmentsByDate(date: Date) List~Appointment~
        +getAppointmentsByCustomer(customerId: int) List~Appointment~
        +getAllAppointments() List~Appointment~
        +toString() String
    }

    class ServiceRecord {
        -int serviceId
        -int appointmentId
        -int vehicleId
        -Date serviceDate
        -String description
        -double laborCost
        -double partsCost
        -double totalCost
        -Date nextServiceDate
        +createRecord() boolean
        +calculateTotal() double
        +getHistory(vehicleId: int) List~ServiceRecord~
        +getServicesByMonth(month: int, year: int) List~ServiceRecord~
        +getMostExpensiveService() ServiceRecord
        +getAllRecords() List~ServiceRecord~
        +toString() String
    }

    class Bill {
        -int billId
        -int serviceId
        -Date billDate
        -double amount
        -String paymentStatus
        -String paymentMethod
        +generateBill() boolean
        +markPaid(id: int, method: String) boolean
        +getPendingBills() List~Bill~
        +getBillByServiceId(serviceId: int) Bill
        +getAllBills() List~Bill~
        +getTotalRevenue() double
        +toString() String
    }

    %% Relationships with cardinalities
    Customer "1" --> "0..*" Vehicle : owns
    Customer "1" --> "0..*" Appointment : books
    Vehicle "1" --> "0..*" Appointment : scheduled for
    Vehicle "1" --> "0..*" ServiceRecord : has
    Appointment "1" --> "0..1" ServiceRecord : results in
    ServiceRecord "1" --> "1" Bill : generates

Sequence Diagram
sequenceDiagram
    actor Staff
    participant UI as System
(Console UI)
    participant CM as CustomerModule
    participant VM as VehicleModule
    participant AM as AppointmentModule
    participant SM as ServiceModule
    participant BM as BillingModule
    participant DB as Database
(MySQL)

    Note over Staff, DB: === Phase 1: Customer Registration ===

    Staff ->> UI: Select "Customer Management"
    UI ->> Staff: Display Customer Sub-Menu

    alt New Customer
        Staff ->> UI: Select "Add Customer"
        UI ->> Staff: Prompt for customer details
        Staff ->> UI: Enter name, phone, email, address
        UI ->> CM: addCustomer(details)
        CM ->> DB: INSERT INTO Customer VALUES(...)
        DB -->> CM: customer_id (auto-generated)
        CM -->> UI: Customer registered (ID: xxx)
        UI -->> Staff: Display confirmation
    else Existing Customer
        Staff ->> UI: Select "Search Customer"
        Staff ->> UI: Enter customer ID
        UI ->> CM: searchCustomer(id)
        CM ->> DB: SELECT * FROM Customer WHERE customer_id = ?
        DB -->> CM: Customer record
        CM -->> UI: Customer data
        UI -->> Staff: Display customer details
    end

    Note over Staff, DB: === Phase 2: Vehicle Registration ===

    Staff ->> UI: Select "Vehicle Management"
    UI ->> Staff: Display Vehicle Sub-Menu
    Staff ->> UI: Select "Add Vehicle"
    UI ->> Staff: Prompt for vehicle details
    Staff ->> UI: Enter reg. no, brand, model, type, year
    UI ->> VM: addVehicle(customerId, details)
    VM ->> DB: INSERT INTO Vehicle VALUES(...)
    DB -->> VM: vehicle_id (auto-generated)
    VM -->> UI: Vehicle registered (ID: xxx)
    UI -->> Staff: Display confirmation

    Note over Staff, DB: === Phase 3: Appointment Scheduling ===

    Staff ->> UI: Select "Appointment Management"
    UI ->> Staff: Display Appointment Sub-Menu
    Staff ->> UI: Select "Book Appointment"
    UI ->> Staff: Prompt for appointment details
    Staff ->> UI: Enter date, issue description
    UI ->> AM: bookAppointment(custId, vehId, date, issue)
    AM ->> DB: INSERT INTO Appointment VALUES(..., status='BOOKED')
    DB -->> AM: appointment_id (auto-generated)
    AM -->> UI: Appointment booked (ID: xxx)
    UI -->> Staff: Display booking confirmation

    Note over Staff, DB: === Phase 4: Service Recording ===

    Staff ->> UI: Select "Service Record Management"
    UI ->> Staff: Display Service Sub-Menu
    Staff ->> UI: Select "Add Service Record"
    UI ->> Staff: Prompt for service details
    Staff ->> UI: Enter description, labor cost, parts cost

    UI ->> SM: createRecord(apptId, vehId, desc, laborCost, partsCost)
    SM ->> SM: calculateTotal(laborCost, partsCost)
    Note right of SM: totalCost = laborCost + partsCost
    SM ->> DB: INSERT INTO Service_Record VALUES(...)
    DB -->> SM: service_id (auto-generated)

    SM ->> AM: updateStatus(apptId, "COMPLETED")
    AM ->> DB: UPDATE Appointment SET status='COMPLETED' WHERE appointment_id = ?
    DB -->> AM: Updated successfully
    AM -->> SM: Status updated

    SM -->> UI: Service recorded (ID: xxx, Total: ₹xxxx)
    UI -->> Staff: Display service record summary

    Note over Staff, DB: === Phase 5: Bill Generation ===

    Staff ->> UI: Select "Billing Management"
    UI ->> Staff: Display Billing Sub-Menu
    Staff ->> UI: Select "Generate Bill"
    UI ->> Staff: Prompt for service ID

    UI ->> BM: generateBill(serviceId)
    BM ->> DB: SELECT total_cost FROM Service_Record WHERE service_id = ?
    DB -->> BM: total_cost value
    BM ->> DB: INSERT INTO Bill VALUES(..., status='PENDING')
    DB -->> BM: bill_id (auto-generated)
    BM -->> UI: Bill generated (ID: xxx, Amount: ₹xxxx)
    UI -->> Staff: Display bill details

    opt Payment Received
        Staff ->> UI: Select "Mark Payment"
        Staff ->> UI: Enter bill ID, payment method
        UI ->> BM: markPaid(billId, paymentMethod)
        BM ->> DB: UPDATE Bill SET payment_status='PAID', payment_method=? WHERE bill_id=?
        DB -->> BM: Updated successfully
        BM -->> UI: Payment recorded
        UI -->> Staff: Display payment confirmation
    end

    Note over Staff, DB: === Phase 6: Reports (Optional) ===

    opt View Reports
        Staff ->> UI: Select "Reports"
        UI ->> Staff: Display Reports Sub-Menu
        Staff ->> UI: Select report type

        UI ->> SM: getHistory(vehicleId)
        SM ->> DB: SELECT * FROM Service_Record WHERE vehicle_id = ? ORDER BY service_date
        DB -->> SM: Service history records
        SM -->> UI: Formatted service history
        UI -->> Staff: Display report
    end
